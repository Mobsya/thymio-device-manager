cmake_minimum_required(VERSION 3.25)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION.txt" TDM_VERSION LIMIT_COUNT 1)
project(ThymioDeviceManager VERSION ${TDM_VERSION} LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Apple Clang adds /usr/local/include as a normal include directory. Pinned
# imported dependencies must precede it rather than being searched as system headers.
set(CMAKE_NO_SYSTEM_FROM_IMPORTED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(BUILD_SHARED_LIBS OFF)
include(CTest)
find_package(Threads REQUIRED)
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)

set(TDM_DOWNLOAD_DIR "${CMAKE_SOURCE_DIR}/.cache/downloads" CACHE PATH "Verified dependency downloads")
set(TDM_DEPS_PREFIX "" CACHE PATH "Architecture-specific dependency installation")
set(TDM_FLATC "" CACHE FILEPATH "Native host flatc executable")
if(NOT EXISTS "${TDM_DEPS_PREFIX}/include/boost/version.hpp" OR NOT EXISTS "${TDM_FLATC}")
  message(FATAL_ERROR "Run make deps (or python scripts/build.py deps --preset <preset>) first.")
endif()
list(PREPEND CMAKE_PREFIX_PATH "${TDM_DEPS_PREFIX}")
file(READ "${CMAKE_SOURCE_DIR}/dependencies.lock.json" dependency_manifest)
string(JSON boost_version GET "${dependency_manifest}" dependencies boost revision)
string(JSON openssl_version GET "${dependency_manifest}" dependencies openssl revision)
set(Boost_USE_STATIC_LIBS ON)
if(WIN32)
  set(Boost_USE_STATIC_RUNTIME ON)
  set(OPENSSL_MSVC_STATIC_RT TRUE)
endif()
find_package(Boost ${boost_version} EXACT CONFIG REQUIRED COMPONENTS chrono date_time filesystem program_options regex thread atomic)
set(OPENSSL_USE_STATIC_LIBS TRUE)
set(OPENSSL_ROOT_DIR "${TDM_DEPS_PREFIX}")
find_package(OpenSSL ${openssl_version} EXACT REQUIRED)

add_library(aseba_conf INTERFACE)
target_include_directories(aseba_conf INTERFACE "${CMAKE_SOURCE_DIR}" "${CMAKE_SOURCE_DIR}/aseba")
target_compile_definitions(aseba_conf INTERFACE ASEBA_ASSERT BOOST_ALL_NO_LIB BOOST_ALLOW_DEPRECATED_HEADERS
  BOOST_THREAD_PROVIDES_FUTURE_CONTINUATION BOOST_BEAST_ALLOW_DEPRECATED)
if(MSVC)
  target_compile_options(aseba_conf INTERFACE /utf-8 /permissive- /bigobj)
  target_compile_definitions(aseba_conf INTERFACE WIN32_LEAN_AND_MEAN NOMINMAX BOOST_USE_WINDOWS_H
    _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS _USE_MATH_DEFINES _WIN32_WINNT=0x0A00 UNICODE _UNICODE)
elseif(APPLE)
  target_compile_definitions(aseba_conf INTERFACE _LIBCPP_ENABLE_CXX17_REMOVED_UNARY_BINARY_FUNCTION)
  target_compile_options(aseba_conf INTERFACE -Wno-enum-constexpr-conversion -fno-aligned-allocation)
endif()

include(cmake/Dependencies.cmake)
set(ASEBA_VERSION "2.4.0")
set(GIT_REVISION "standalone-${PROJECT_VERSION}")
set(LIB_VERSION_STRING "3.0.0")
add_subdirectory(aseba/common)
add_subdirectory(aseba/compiler)
add_subdirectory(aseba/flatbuffers)
add_subdirectory(aseba/thymio-device-manager)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
